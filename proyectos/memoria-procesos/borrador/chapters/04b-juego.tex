% Corrección de los diagramas de secuencia del capítulo del juego:
% dentro de \begin{call}...\end{call} las llamadas anidadas van con
% \begin{call}...\end{call}, no con el comando \call. En \mess no hay
% paréntesis y en los textos no hay _.

% --- fig:sec-partida ---
\begin{figure}[htbp]
\centering
\resizebox{\textwidth}{!}{%
\begin{sequencediagram}
  \newthread{j}{Jugador}
  \newinst[2]{g}{juego.js en navegador}
  \newinst[2]{sim}{mundo.js simulacion}
  \newinst[2]{cli}{api.js}
  \newinst[2]{api}{API /api/juego}
  \newinst[2]{srv}{juegoService}
  \newinst[2]{repo}{juegoRepository}
  \begin{call}{j}{jugar con jugador y entrenador}{g}{ok}
    \begin{call}{g}{iniciar partida}{cli}{ok}
    \end{call}
    \begin{call}{cli}{POST partidas}{api}{201}
    \end{call}
    \begin{call}{api}{iniciarPartida}{srv}{partida}
    \end{call}
    \begin{call}{srv}{lee perfil y version}{repo}{perfil}
    \end{call}
    \begin{call}{srv}{guarda partida en curso}{repo}{ok}
    \end{call}
  \end{call}
  \begin{sdblock}{loop}{cada fotograma con paso fijo}
    \begin{call}{g}{avanzar con entrada}{sim}{eventos}
    \end{call}
    \mess{sim}{estado fin, te eliminan}{g}
  \end{sdblock}
  \begin{call}{g}{terminar con estadisticas}{cli}{ok}
    \begin{call}{cli}{POST partidas id fin}{api}{200 o 400}
    \end{call}
    \begin{call}{api}{finalizarPartida}{srv}{validacion}
    \end{call}
    \begin{sdblock}{alt}{resultado imposible}
      \mess{srv}{400 resultado no creible}{api}
    \end{sdblock}
    \begin{sdblock}{else}{resultado valido}
      \begin{call}{srv}{reparte recompensas y guarda}{repo}{perfil}
      \end{call}
      \mess{api}{200 con recompensas y logros}{g}
    \end{sdblock}
  \end{call}
\end{sequencediagram}}
\caption{Una partida del juego con validación en el servidor.}
\label{fig:sec-partida}
\end{figure}
